	\begin{frame}{Question}
    Construct a pair of tangents to a circle of radius \(4\,\text{cm}\) from a point \(P\) at a distance of \(7\,\text{cm}\) from its centre \(O\).
\end{frame}
\begin{frame}{Solution}
    Let the equation of the circle with centre $\vec{u}$ be
\begin{align}
(\vec{x} - \vec{u})^\top (\vec{x} - \vec{u}) = r^2,
\end{align}

Given,
\begin{align}
r = 4, \quad \|\vec{P} - \vec{u}\| = 7.
\end{align}

Let
\begin{align}
\vec{u}=\vec{0},\quad
\vec{v} = \vec{P} - \vec{u}, \quad
s = \vec{v}^\top \vec{v} = \|\vec{v}\|^2.
\end{align}

Define the orthonormal basis vectors as
\begin{align}
\vec{v_1} = \frac{\vec{v}}{\sqrt{s}}, \qquad
\vec{v_2} = \frac{\vec{Jv}}{\sqrt{s}},\quad
\vec{J} = \myvec{0 & -1 \\ 1 & 0}.
\end{align}

Let the unit normal to the tangent at the point of contact be expressed as
\begin{align}
\frac{\vec{n}}{\|\vec{n}\|} = \alpha \vec{v_1} + \beta \vec{v_2}. \label{eq:nexpand}
\end{align}
\end{frame}
\begin{frame}{Solution}
The points of contact on the circle are given by
\begin{align}
\vec{q_i} = \vec{u} + r\,\frac{\vec{n_i}}{\|\vec{n_i}\|}, \qquad i = 1,2.
\end{align}
Tangency condition:  \\

For tangents from an external point \(\vec{P}\) to the circle, we have
\begin{align}
\vec{v}^\top \frac{\vec{n}}{\|\vec{n}\|} = r. \label{eq:tangency}
\end{align}

Substituting \eqref{eq:nexpand} into \eqref{eq:tangency}, and noting that
\(\vec{v} = \sqrt{s}\,\vec{v_1}\) and \(\vec{v}^\top \vec{v_2} = 0\), we get
\begin{align}
\alpha \sqrt{s} = r
\quad \Rightarrow \quad
\boxed{\alpha = \frac{r}{\sqrt{s}}.} \label{eq:alpha}
\end{align}

 Unit length condition:
\begin{align}
\alpha^2 + \beta^2 &= 1. \label{eq:unit}
\end{align}
\end{frame}
\begin{frame}{Solution}
\begin{align}
\beta^2 &= 1 - \frac{r^2}{s}
\quad \Rightarrow \quad
\boxed{\beta = \pm\sqrt{1 - \frac{r^2}{s}}.} \label{eq:beta}
\end{align}

Hence,
\begin{align}
\boxed{
\frac{\vec{n}}{\|\vec{n}\|}
=
\frac{r}{\sqrt{s}}\,\vec{v_1}
\;\pm\;
\sqrt{1 - \frac{r^2}{s}}\,\vec{v_2}.
} \label{eq:finaln}
\end{align}

\begin{align}
\Rightarrow\;
\boxed{
\vec{q_i}
=
\vec{u}
+
\frac{r^2}{\sqrt{s}}\,\vec{v_1}
\;\pm\;
r\sqrt{1 - \frac{r^2}{s}}\,\vec{v_2}.
}
\end{align}
\end{frame}
\begin{frame}{Solution}
Given \(r=4,\ \sqrt{s}=7,\vec{u}=\vec{0}\),
\begin{align}
\frac{r^2}{\sqrt{s}} &= \frac{16}{7} \\
r\sqrt{1 - \frac{r^2}{s}} &= 4\sqrt{1 - \frac{16}{49}} = \frac{4\sqrt{33}}{7}.
\end{align}
\begin{align}
\vec{q_1} = \myvec{\dfrac{16}{7}\\[4pt]\dfrac{4\sqrt{33}}{7}},\quad
\vec{q_2} = \myvec{\dfrac{16}{7}\\[4pt]-\dfrac{4\sqrt{33}}{7}}.
\end{align}

\end{frame}

 #include <stdio.h>
#include <math.h>

// Tangent function to be called from Python
void tangent_points(double r, double px, double py, double *Ax, double *Ay, double *Bx, double *By) {
    double ox = 0.0, oy = 0.0;
    double d = sqrt((px - ox)*(px - ox) + (py - oy)*(py - oy));

    if (d <= r) {
        *Ax = *Ay = *Bx = *By = NAN;
        return;
    }

    double r2 = r*r;
    double d2 = d*d;
    double k = r2 / d2;
    double h = (r / d2) * sqrt(d2 - r2);
    \end{lstlisting}
\end{frame}
\begin{frame}[fragile]

    *Ay = k*py + h*px;
    *Bx = k*px + h*py;
    *By = k*py - h*px;
}

import numpy as np
import matplotlib.pyplot as plt
from numpy import linalg as LA

# Local imports
from libs.line.funcs import line_gen
from libs.conics.funcs import circ_gen

# --- Circle and external point ---
O = np.array([[0.0], [0.0]])     # Centre
r = 4.0                          # Radius
P = np.array([[7.0], [0.0]])     # External point

# --- Tangent point calculation ---
u = P - O
d = float(LA.norm(u))
u_hat = u / d

# Rotation matrix for 90 degrees
R90 = np.array([[0, -1], [1, 0]])

# Tangent points formula
r2 = r**2
sqrt_term = np.sqrt(d**2 - r**2)

A = O + (r2/d)*u_hat + (r/d)*sqrt_term*(R90 @ u_hat)
B = O + (r2/d)*u_hat - (r/d)*sqrt_term*(R90 @ u_hat)

# --- Generate required curves ---
x_circ = circ_gen(O, r)
x_tan1 = line_gen(P, A)
x_tan2 = line_gen(P, B)
x_OP = line_gen(O, P)
