	\begin{frame}{Question}
    Construct a pair of tangents to a circle of radius \(4\,\text{cm}\) from a point \(P\) at a distance of \(7\,\text{cm}\) from its centre \(O\).
\end{frame}
\begin{frame}{Solution}
 The tangent directions \(\vec{m}\) from an external point \(\vec{h}\) to the circle 
\begin{align}
g(\vec{x}) = \vec{x}^\top \vec{x} - r^2 = 0
\end{align}

satisfy 
\begin{align}
\vec{m}^\top \vec{\Phi} \vec{m} = 0
\end{align}

where 
\begin{align}
\vec{\Phi} &= \vec{h}\vec{h}^\top - g(\vec{h})\vec{I} \label{eq:sigma_def}
\end{align}

With the point \(\vec{h} = d\vec{e}_1\), we have \(g(\vec{h}) = d^2 - r^2\). From \eqref{eq:sigma_def},
\begin{align}
\vec{\Phi} &= 
\myvec{
d^2 & 0 \\
0 & 0
}
-
(d^2 - r^2)
\myvec{
1 & 0 \\
0 & 1
}
\\
&=
\myvec{
r^2 & 0 \\
0 & -(d^2 - r^2)
} \label{eq:sigma_matrix}
\end{align}

Since \(\vec{\Phi}\) is a diagonal matrix, its eigenvalues are:
\begin{align}
\lambda_1 &= r^2, \quad \lambda_2 = -(d^2 - r^2) \label{eq:eigenvalues}
\end{align}
\end{frame}
\begin{frame}{Solution}
The matrix of orthonormal eigenvectors is 
\[
\vec{P} = \vec{I}
\]

Hence, the unit direction vectors of the tangents are given by:
\begin{align}
\vec{m} = \frac{1}{\sqrt{\lambda_1 - \lambda_2}} \, \vec{P} 
\myvec{
\sqrt{-\lambda_2} \\[4pt]
\pm \sqrt{\lambda_1}
}
\label{eq:m_general}
\end{align}

From \eqref{eq:eigenvalues},
\begin{align}
\lambda_1 - \lambda_2 &= r^2 - (-(d^2 - r^2)) = d^2 \\
-\lambda_2 &= d^2 - r^2
\end{align}

Substituting into \eqref{eq:m_general}:
\begin{align}
\vec{m} 
&= \frac{1}{\sqrt{d^2}} 
\myvec{
\sqrt{d^2 - r^2} \\[4pt]
\pm r
}
\\
&= \frac{1}{d}
\myvec{
\sqrt{d^2 - r^2} \\[4pt]
\pm r
}
\label{eq:m_final}
\end{align}
\end{frame}
\begin{frame}{Solution}
The points of contact are given by 
\begin{align}
\vec{q} = \vec{h} + \kappa \vec{m}
\end{align}

Satisfies the circle equation
\begin{align}
g(\vec{h} + \kappa\vec{m}) 
&= (\vec{h} + \kappa\vec{m})^\top (\vec{h} + \kappa\vec{m}) - r^2 = 0 \\
&\implies 
\kappa^2 (\vec{m}^\top \vec{m}) + 2\kappa(\vec{h}^\top \vec{m}) + g(\vec{h}) = 0
\end{align}

For a tangent, \(\kappa\) has a single repeated root.
Since \(\vec{m}\) is a unit vector (\(\vec{m}^\top \vec{m} = 1\)):
\begin{align}
\kappa &= -(\vec{h}^\top \vec{m}) \label{eq:kappa_def}
\end{align}

\begin{align}
\vec{h}^\top \vec{m} 
&= (d\vec{e}_1)^\top 
\frac{1}{d}
\myvec{
\sqrt{d^2 - r^2} \\[4pt]
\pm r
}
= \sqrt{d^2 - r^2} 
\end{align}

Hence,
\[
\kappa = -\sqrt{d^2 - r^2}
\]
\end{frame}
\begin{frame}{Solution}
The points of contact are:
\begin{align}
\vec{q} 
&= d\vec{e}_1 - \sqrt{d^2 - r^2}
\frac{1}{d}
\myvec{
\sqrt{d^2 - r^2} \\[4pt]
\pm r
}
\\[6pt]
&=
\myvec{
\dfrac{r^2}{d} \\[6pt]
\mp \dfrac{r\sqrt{d^2 - r^2}}{d}
}
\label{eq:points}
\end{align}

\textbf{Substituting \(r = 4, \; d = 7\):}
\begin{align}
\sqrt{d^2 - r^2} &= \sqrt{49 - 16} = \sqrt{33} \\
\vec{q} 
&=
\myvec{
\dfrac{16}{7} \\[6pt]
\mp \dfrac{4\sqrt{33}}{7}
}
\end{align}
\end{frame}
\begin{frame}{Solution}
Thus, the points of contact of the tangents are:
\[
\boxed{
\vec{q}_1 = 
\myvec{
\dfrac{16}{7} \\[4pt]
\dfrac{4\sqrt{33}}{7}
}, 
\quad
\vec{q}_2 =
\myvec{
\dfrac{16}{7} \\[4pt]
-\dfrac{4\sqrt{33}}{7}
}
}
\]
\end{frame}

 #include <stdio.h>
#include <math.h>

// Tangent function to be called from Python
void tangent_points(double r, double px, double py, double *Ax, double *Ay, double *Bx, double *By) {
    double ox = 0.0, oy = 0.0;
    double d = sqrt((px - ox)*(px - ox) + (py - oy)*(py - oy));

    if (d <= r) {
        *Ax = *Ay = *Bx = *By = NAN;
        return;
    }

    double r2 = r*r;
    double d2 = d*d;
    double k = r2 / d2;
    double h = (r / d2) * sqrt(d2 - r2);
    \end{lstlisting}
\end{frame}
\begin{frame}[fragile]

    *Ay = k*py + h*px;
    *Bx = k*px + h*py;
    *By = k*py - h*px;
}

import numpy as np
import matplotlib.pyplot as plt
from numpy import linalg as LA

# Local imports
from libs.line.funcs import line_gen
from libs.conics.funcs import circ_gen

# --- Circle and external point ---
O = np.array([[0.0], [0.0]])     # Centre
r = 4.0                          # Radius
P = np.array([[7.0], [0.0]])     # External point

# --- Tangent point calculation ---
u = P - O
d = float(LA.norm(u))
u_hat = u / d

# Rotation matrix for 90 degrees
R90 = np.array([[0, -1], [1, 0]])

# Tangent points formula
r2 = r**2
sqrt_term = np.sqrt(d**2 - r**2)

A = O + (r2/d)*u_hat + (r/d)*sqrt_term*(R90 @ u_hat)
B = O + (r2/d)*u_hat - (r/d)*sqrt_term*(R90 @ u_hat)

# --- Generate required curves ---
x_circ = circ_gen(O, r)
x_tan1 = line_gen(P, A)
x_tan2 = line_gen(P, B)
x_OP = line_gen(O, P)
