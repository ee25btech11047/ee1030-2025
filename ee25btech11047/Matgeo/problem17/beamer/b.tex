	\begin{frame}{Question}
    Construct a pair of tangents to a circle of radius \(4\,\text{cm}\) from a point \(P\) at a distance of \(7\,\text{cm}\) from its centre \(O\).
\end{frame}
\begin{frame}{Solution}
   Given:\\

Let the centre be origin and P lies on x axis. 
\begin{align}
\vec{x}^T\vec{x} = 16 \label{eq1} \\
\vec{P} = \myvec{7\\0}, 
\end{align}

Equation of a line through  $\vec{P}$  with direction  $\vec{d}$:
\begin{align}
&\vec{x} = \vec{P} + t\vec{d}, 
\quad t\in\mathbb{R}.
\label{eq2}
\end{align}
\begin{align}
\text{Substitute} \quad \eqref{eq2} in \eqref{eq1}:
(\vec{P}+t\vec{d})^T(\vec{P}+t\vec{d}) = 16 \nonumber\\
\Rightarrow 
(\vec{d}^T\vec{d})t^2 
+ 2(\vec{P}^T\vec{d})t 
+ (\vec{P}^T\vec{P} - 16) = 0
\label{eq3}
\end{align}

For tangency, discriminant = 0: 
\begin{align}
(2\vec{P}^T\vec{d})^2 
- 4(\vec{d}^T\vec{d})(\vec{P}^T\vec{P} - 16) = 0 \nonumber\\
\Rightarrow 
(\vec{P}^T\vec{d})^2 
= (\vec{d}^T\vec{d})(\vec{P}^T\vec{P} - 16)
\label{eq4}
\end{align}
\end{frame}
\begin{frame}{Solution}
\begin{align}
\vec{P}^T\vec{P} &= 49, \\
(\vec{P}^T\vec{d})^2 &= 33(\vec{d}^T\vec{d}).\\
\text{For a unit vector } \vec{d},\nonumber\\ 
\ \vec{d}^T\vec{d} &= 1 
\Rightarrow 
(\vec{P}^T\vec{d})^2 = 33
\label{eq5}
\end{align}

Let
\begin{align}
\vec{M} = \vec{P}\vec{P}^T\implies 
\vec{d}^T\vec{M}\vec{d} &= (\vec{P}^T\vec{d})^2 \\ 
\vec{d}^T\vec{M}\vec{d} = 33, 
\label{eq6}
\end{align}

\begin{align}
\vec{M} 
= \vec{P}\vec{P}^T 
= 
\myvec{7\\0}\myvec{7 & 0}
= 
\myvec{49 & 0\\0 & 0}
\label{eq7}
\end{align}
\end{frame}
\begin{frame}{Solution}
Eigen values and Eigen vectors of M:
\begin{align}
\lambda_1 &= 49, 
\ \lambda_2 = 0. \nonumber\\
\vec{e}_1 &= \myvec{1\\0}, 
\quad 
\vec{e}_2 = \myvec{0\\1}
\label{eq8}
\end{align}

$\vec{d}$ in orthonormal Eigenbasis : \quad $\vec{d}^T\vec{d} = 1$
 \begin{align}
\vec{d} 
= \alpha\vec{e}_1 + \beta\vec{e}_2, 
\quad 
\alpha^2 + \beta^2 = 1
\label{eq9}\\
\vec{d}^T\vec{M}\vec{d} 
= \lambda_1\alpha^2 + \lambda_2\beta^2 \\
\Rightarrow 
\vec{d}^T\vec{M}\vec{d} 
= 49\alpha^2 + 0\cdot\beta^2 
= 49\alpha^2
\label{eq10}\\
From \quad \eqref{eq6}, 
49\alpha^2 = 33 
\Rightarrow 
\alpha &= \pm\frac{\sqrt{33}}{7}. \\
\Rightarrow 
\beta = \pm\frac{4}{7}
\label{eq11}
\end{align}
\end{frame}
\begin{frame}{Solution}
Hence $\vec{d}$ is:
\begin{align}
&\therefore 
\vec{d}
= 
\myvec{\frac{\sqrt{33}}{7}\\[3pt]\pm\frac{4}{7}}
\label{eq12}
\end{align}

From $\eqref{eq3}$ and $\eqref{eq5}$
\begin{align}
t &= -\frac{\vec{P}^T\vec{d}}{\vec{d}^T\vec{d}}. \nonumber\\
\Rightarrow 
t &= -\sqrt{33}
\label{eq13}
\end{align}

Required Points of Contact
\begin{align}
\vec{T} 
&= \vec{P} + t\vec{d} 
= \vec{P} - \sqrt{33}\vec{d}
\label{eq14}\\
\vec{P} &= \myvec{7\\0}, 
 \vec{d} = 
\myvec{\frac{\sqrt{33}}{7}\\[3pt]\pm\frac{4}{7}}: \\
\vec{T} 
&= 
\myvec{7\\0} 
- \sqrt{33}
\myvec{\frac{\sqrt{33}}{7}\\[3pt]\pm\frac{4}{7}} 
= 
\myvec{\frac{16}{7}\\[3pt]\mp\frac{4\sqrt{33}}{7}}
\label{eq15}
\end{align}
\end{frame}
\begin{frame}{Solution}
    \begin{align}
\vec{T}_1 &= \myvec{\frac{16}{7}\\[3pt]\frac{4\sqrt{33}}{7}}, 
\quad 
\vec{T}_2 = \myvec{\frac{16}{7}\\[3pt]-\frac{4\sqrt{33}}{7}}
\label{eq16}
\end{align}


Equation of tangents :
\begin{align}
\vec{x}= \myvec{7 & 0}^T +k\myvec{\frac{\sqrt{33}}{7}&\frac{4}{7}}^T\\
\vec{x}= \myvec{7 & 0}^T +k\myvec{\frac{\sqrt{33}}{7}&-\frac{4}{7}}^T
\end{align}

\end{frame}

 #include <stdio.h>
#include <math.h>

// Tangent function to be called from Python
void tangent_points(double r, double px, double py, double *Ax, double *Ay, double *Bx, double *By) {
    double ox = 0.0, oy = 0.0;
    double d = sqrt((px - ox)*(px - ox) + (py - oy)*(py - oy));

    if (d <= r) {
        *Ax = *Ay = *Bx = *By = NAN;
        return;
    }

    double r2 = r*r;
    double d2 = d*d;
    double k = r2 / d2;
    double h = (r / d2) * sqrt(d2 - r2);
    \end{lstlisting}
\end{frame}
\begin{frame}[fragile]

    *Ax = k*px - h*py;
    *Ay = k*py + h*px;
    *Bx = k*px + h*py;
    *By = k*py - h*px;
}

import numpy as np
import matplotlib.pyplot as plt
from numpy import linalg as LA

# Local imports
from libs.line.funcs import line_gen
from libs.conics.funcs import circ_gen

# --- Circle and external point ---
O = np.array([[0.0], [0.0]])     # Centre
r = 4.0                          # Radius
P = np.array([[7.0], [0.0]])     # External point

# --- Tangent point calculation ---
u = P - O
d = float(LA.norm(u))
u_hat = u / d
\end{lstlisting}

# Rotation matrix for 90 degrees
R90 = np.array([[0, -1], [1, 0]])

# Tangent points formula
r2 = r**2
sqrt_term = np.sqrt(d**2 - r**2)

A = O + (r2/d)*u_hat + (r/d)*sqrt_term*(R90 @ u_hat)
B = O + (r2/d)*u_hat - (r/d)*sqrt_term*(R90 @ u_hat)

# --- Generate required curves ---
x_circ = circ_gen(O, r)
x_tan1 = line_gen(P, A)
x_tan2 = line_gen(P, B)
x_OP = line_gen(O, P)

# --- Plotting ---
plt.plot(x_circ[0, :], x_circ[1, :] )
