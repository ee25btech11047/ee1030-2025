	\begin{frame}{Question}
    Find the area of the region bounded by the parabola $y^2 = x$ and the line $2y = x$.
\end{frame}
\begin{frame}{Solution}
    \begin{align}
\vec{x}^T \vec{V} \vec{x} + 2\vec{u}^T\vec{x} + f = 0 \\
\vec{V} = \myvec{0 & 0 \\ 0 & 1},\quad 
\vec{u} = \myvec{-\tfrac{1}{2} \\ 0},\quad f=0 \\
\end{align}
\begin{align}
\vec{x} &= \vec{h} + \kappa \vec{m}, \quad \kappa \in \mathbb{R} \\
\vec{h} &= \myvec{0 \\ 0}, \quad 
\vec{m} = \myvec{2 \\ 1} \\
\vec{x}_i &= \vec{h} + \kappa_i \vec{m} 
\end{align}

Substituting eq-(6) in eq-(1) and Solving the Quadratic 
\begin{align}
\kappa_i = \frac{1}{\vec{m}^T \vec{V} \vec{m}}
\brak{-\vec{m}^T(\vec{V}\vec{h}+\vec{u})
\pm \sqrt{(\vec{m}^T(\vec{V}\vec{h}+\vec{u}))^2 - g(\vec{h})(\vec{m}^T \vec{V} \vec{m})}} 
\end{align}
\end{frame}
\begin{frame}{Solution}
\begin{align}
g(\vec{h}) &= \vec{h}^T \vec{V} \vec{h} + 2\vec{u}^T \vec{h} + f\\
\vec{V}\vec{m} &= \myvec{0 \\ 1},\quad 
\vec{m}^T \vec{V} \vec{m} = 1 \\
\vec{V}\vec{h}+\vec{u} &= \myvec{-\tfrac{1}{2} \\ 0},\quad
\vec{m}^T(\vec{V}\vec{h}+\vec{u}) = -1 \\
g(\vec{h}) &= 0\\
\kappa_i &= -(-1) \pm \sqrt{(-1)^2 - 0} 
= 1 \pm 1 \\
\implies \kappa_1 &= 0,\;\; \kappa_2 = 2\\
\vec{x}_1 &= \vec{h} + 0\vec{m} = \myvec{0 \\ 0},\quad
\vec{x}_2 = \vec{h} + 2\vec{m} = \myvec{4 \\ 2} \\
\text{Area} &= \int_{0}^{2} \Big(2y - y^2\Big)\,dy 
= \brak{y^2 - \tfrac{y^3}{3}}_0^2 
= 4 - \tfrac{8}{3} 
= \tfrac{4}{3}
\end{align}
\end{frame}

double g(double h[2], double V[2][2], double u[2], double f) {
    double temp1 = h[0]*(V[0][0]*h[0] + V[0][1]*h[1]) + 
                   h[1]*(V[1][0]*h[0] + V[1][1]*h[1]);
    double temp2 = 2*(u[0]*h[0] + u[1]*h[1]);
    return temp1 + temp2 + f;
}
