	\begin{frame}{Question}
    If the sum of zeroes of the polynomial is $\sqrt{2}$, then find the value of $k$ .
\begin{align*}
    p(x)=2x^2-k\sqrt{2}x+1
\end{align*}

\end{frame}
\begin{frame}{Solution}
    \begin{align}
\vec{x}^\top \vec{V} \vec{x} + 2\vec{u}^\top \vec{x} + f = 0\\
\vec{x} = \vec{h} + \kappa \vec{m}.
\end{align}

Substituting the line into the conic gives
\begin{align}
(\vec{h} + \kappa \vec{m})^\top \vec{V} (\vec{h} + \kappa \vec{m}) + 2\vec{u}^\top (\vec{h} + \kappa \vec{m}) + f &= 0 \\
\Rightarrow\;
\kappa^2(\vec{m}^\top \vec{V} \vec{m}) + 2\kappa\, \vec{m}^\top(\vec{V}\vec{h} + \vec{u}) + (\vec{h}^\top \vec{V} \vec{h} + 2\vec{u}^\top \vec{h} + f) &= 0\\
\kappa_1 + \kappa_2 = -\frac{2\,\vec{m}^\top(\vec{V}\vec{h}+\vec{u})}{\vec{m}^\top \vec{V} \vec{m}}.
\end{align}

Given:(Roots are given by intersection of conic with x-axis)
\begin{align}
\vec{V} &=
\myvec{
2 & 0\\[4pt]
0 & 0
}, \quad
\vec{u} =
\myvec{
-\dfrac{k\sqrt{2}}{2}\\[4pt]
0
}, \quad
f = 1\\
\vec{h}&=\myvec{0 \\ 0},\quad \vec{m}=\myvec{1\\0}\\
\end{align}
\end{frame}
\begin{frame}{Solution}
\begin{align}
\vec{V}\vec{h} + \vec{u} &=
\myvec{
- \dfrac{k\sqrt{2}}{2}\\[4pt]
0
},\\[4pt]
\vec{m}^\top \vec{V} \vec{m} &= 2.\\
\vec{m}^\top (\vec{V}\vec{h} + \vec{u})&= - \dfrac{k\sqrt{2}}{2}
\end{align}

\begin{align}
\kappa_1 + \kappa_2 
&= -\frac{2(- \dfrac{k\sqrt{2}}{2})}{2}=\sqrt{2}.
\end{align}
\begin{align}
\Rightarrow\;\boxed{k = 2}.
\end{align}

\end{frame}

    // κ₁ + κ₂ = -[2 * (mᵀ(Vh + u))] / (mᵀ V m)

    // mᵀ V m = 2,  (Vh + u) = [-k√2/2, 0]ᵀ

    Vh_plus_u[1] = V[1][0]*h[0] + V[1][1]*h[1] + u[1];

    double mT_V_m = m[0]*(V[0][0]*m[0] + V[0][1]*m[1]) +
