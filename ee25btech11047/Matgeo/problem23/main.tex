\textbf{Question:} \\

If
\begin{align*}
\vec{A} \vec{B} = \vec{I},
\end{align*}

then:

\begin{multicols}{4}
\begin{enumerate}[label=\alph*)]
    \item $\vec{B} = \vec{A}^{\top}$
    \item $\vec{A} = \vec{B}^{\top}$
    \item $\vec{B} = \vec{A}^{-1}$
    \item $\vec{B} = \vec{A}$
\end{enumerate}
\end{multicols}

\textbf{Solution:}\\

Given:
\begin{align}
 \vec{A} \vec{B} = \vec{I} 
 \end{align}
 
Multiply both sides by  $\vec{A}^{-1}$: 
\begin{align}
&\vec{A}^{-1} (\vec{A} \vec{B}) = \vec{A}^{-1} \vec{I} \\
\Rightarrow \quad & (\vec{A}^{-1} \vec{A}) \vec{B} = \vec{A}^{-1} \\
\Rightarrow \quad & \vec{I} \, \vec{B} = \vec{A}^{-1} \\
\Rightarrow \quad & \vec{B} = \vec{A}^{-1}
\end{align}
\begin{align}
 \boxed{\vec{B} = \vec{A}^{-1}}.
\end{align}
